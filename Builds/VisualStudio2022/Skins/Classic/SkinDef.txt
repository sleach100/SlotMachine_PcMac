; SKIN: [Your Skin Name Here]

; =============================================================================
; PANEL / SLOT BACKGROUNDS
; =============================================================================
; Custom background images for each slot panel state
; Images are stretched to fit the entire slot panel area
; =============================================================================

; Background: Normal slot background (default state)
; Displayed when slot is idle or playing
;Background=

; BackgroundFlash: Flash background shown during beat hits
; Displayed momentarily when a beat triggers
;BackgroundFlash=

; BackgroundSampleLoaded: Background shown when a sample is loaded
; Displayed when slot has an audio file loaded (optional variation)
;BackgroundSampleLoaded=


; =============================================================================
; TEXT BOXES / LABELS
; =============================================================================
; Customize appearance of text displays (file names, numeric values, dropdowns)
; =============================================================================

; TextBox: Background image for text labels and combo boxes
; Applied to: File name display, MIDI channel selector, numeric value labels
; Excludes: Mute, Solo, and Master BPM text labels
; Image will be stretched to fit component bounds
;TextBox=

; MIDI_ListPanelBackground: Background for MIDI channel dropdown menu
; Applied to: The popup list that appears when selecting MIDI channels
; Falls back to default if not specified
;MIDI_ListPanelBackground=

; FileNameWidth: Fixed width in pixels for the file name label
; 0 or omitted = auto-width (default ~142px, scales with window)
; Useful to prevent text from extending into decorative bezels
;FileNameWidth=0

; TextBoxFont: Custom font file (.ttf) for numeric displays
; Applied to: Count/Rate, Volume, Decay values, Master BPM slider value
; Must be a TrueType font file in the skin folder
;TextBoxFont=

; TextBoxFontSize: Font size in points for numeric displays
; 0 or omitted = use default size
; Only applied when TextBoxFont is specified
;TextBoxFontSize=0

; TextBoxFontColor: Hex color code for numeric text (e.g., #FFFF33, #F0F)
; Applied to: Numeric value displays where TextBoxFont is used
; Format: #RRGGBB or #RGB (# is optional)
;TextBoxFontColor=

; MidiSelectorColor: Hex color code for MIDI channel selector text
; Applied to: MIDI channel selector, dropdown menu, sample selector
; Overrides TextBoxFontColor if specified
; Format: #RRGGBB or #RGB (# is optional)
;MidiSelectorColor=


; =============================================================================
; BUTTONS
; =============================================================================
; Custom images for button states and appearance
; =============================================================================

; LoadButtonWidth: Width in pixels for the Load button
; 0 or omitted = auto-width (default 110px scaled)
; Right edge is anchored; width changes extend/reduce from left edge
;LoadButtonWidth=0

; Button: Normal button state image
; Applied to: All buttons in normal/idle state
; Image will be stretched to fit button bounds
;Button=

; ButtonHover: Button hover state image
; Displayed when mouse cursor is over the button
; Falls back to Button image if not specified
;ButtonHover=

; ButtonClicked: Button pressed/clicked state image
; Displayed when button is being clicked/held down
; Falls back to Button image if not specified
;ButtonClicked=

; TabSelected: Selected tab button image
; Applied to: Active tab when multiple tabs are present
; Only used when there are 2+ tabs; single tabs use normal Button image
; Falls back to ButtonClicked if not specified
;TabSelected=

; ButtonFontColor: Hex color code for button text
; Applied to: All master row buttons (except Start/Stop), Tabs, Master BPM label,
;             Slot buttons (Load and X), MIDI channel selector, Mute label, Solo label
; Format: #RRGGBB or #RGB (# is optional)
;ButtonFontColor=


; =============================================================================
; SLIDERS (LINEAR / HORIZONTAL)
; =============================================================================
; Customize Master BPM slider and other linear slider controls
; =============================================================================

; SliderThumb: Image for the slider thumb/handle
; Applied to: Master BPM slider thumb and other linear sliders
; Image drawn at scaled size, centered on thumb position
;SliderThumb=

; SliderThumbScale: Scale percentage for slider thumb image
; Range: 1-100 (e.g., 50 = half size, 100 = full size)
; Only applied when SliderThumb is specified
;SliderThumbScale=100

; SliderTrack: Background image for the slider track/rail
; Applied to: Master BPM slider track background
; Scaled/stretched to fit slider width while maintaining aspect ratio
; Only takes effect when SliderThumb is also provided
;SliderTrack=


; =============================================================================
; AUDIO KNOBS (ROTARY SLIDERS)
; =============================================================================
; Customize rotary knobs for Count/Rate, Gain, Decay controls
; Two modes: Filmstrip (multiple frames) or Static (single rotating image)
; =============================================================================

; Knobs: Filmstrip image file or static knob image
; For filmstrip: Contains all rotation frames stacked vertically or horizontally
; For static: Single knob image that will be rotated in real-time
;Knobs=

; KnobFrames: Number of frames in the filmstrip
; Required for filmstrip mode, ignored for static mode
; Example: 50 frames = 50 different rotation positions
;KnobFrames=50

; KnobFilmstripOrientation: Layout direction of filmstrip frames
; Options: Vertical (frames stacked top to bottom)
;          Horizontal (frames arranged left to right)
; Only relevant for filmstrip mode
;KnobFilmstripOrientation=Vertical

; StaticKnob: Enable real-time rotation instead of filmstrip animation
; Options: True (rotate single image), False (use filmstrip frames)
; When True, the Knobs image is rotated dynamically; KnobFrames is ignored
;StaticKnob=False

; KnobShadow: Add a soft shadow beneath static knobs
; Format: radiusPercent,outerEdgeOpacity,verticalOffsetPercent,horizontalOffsetPercent
; radiusPercent: Shadow size as % of knob radius (e.g., 110 = 10% larger)
; outerEdgeOpacity: Shadow opacity at outer edge, 0-100 (e.g., 10 = very soft)
; verticalOffsetPercent: Vertical shadow offset as % of knob size (e.g., 15)
; horizontalOffsetPercent: Horizontal shadow offset as % of knob size (e.g., 15)
; Only visible when StaticKnob=True
; Example: KnobShadow=110,10,15,15
;KnobShadow=


; =============================================================================
; TOGGLES / LED INDICATORS
; =============================================================================

;MuteON=

;MuteOFF=

; SoloON: Image shown when Solo is enabled (channel soloed)
;SoloON=

; SoloOFF: Image shown when Solo is disabled (normal playback)
;SoloOFF=

;ProgressBarWidth=0

;HideProgressBar=False
