; Slot Machine Skin Definition File
; Lines starting with ; are comments and will be ignored

; Knob Settings
Knobs=knob.png
KnobFrames=50
KnobFilmstripOrientation=Vertical

; KnobFilmstripOrientation options:
;   Vertical   - Frames are stacked top to bottom
;   Horizontal - Frames are arranged left to right

; Slot Background Images
; These images replace the default slot frames
; Leave blank to use default JUCE frame rendering
;Background=slot_background.png
;BackgroundFlash=slot_background_flash.png

; Background: Normal slot background image (shown behind sliders, buttons, etc.)
; BackgroundFlash: Flash background image shown when a beat hits
; Images should be sized appropriately for the slot dimensions

; Text Box Images
; Replace the default text label/combobox backgrounds (file name display, MIDI channel selector, etc.)
; Leave blank to use default JUCE rendering
;TextBox=textbox_background.png

; TextBox: Background image for text labels, combo boxes, and popup menus in slots
; Image will be stretched to fit the component bounds, with text/arrow drawn on top
; Applies to: File name labels, MIDI channel selector, dropdown popup menus
; Excludes: Mute and Solo labels (left with default rendering)

; File Name Label Width
; Set a fixed width in pixels for the file name text box
; Leave at 0 or omit for automatic/dynamic width (fills available space)
; Useful for preventing the file name from extending into the bezel
;FileNameWidth=0

; FileNameWidth: Width of the file name label in pixels
; 0 = auto (default: ~142px at 1280px window width, scales dynamically)
; Example: FileNameWidth=120 (sets fixed 120px width)

; Load Button Width
; Set a custom width in pixels for the Load button
; The right edge is anchored - width changes extend/reduce from the left edge
; Leave at 0 or omit for default width (110px scaled)
; Button will not extend past the visible drawing area
;LoadButtonWidth=0

; LoadButtonWidth: Width of the Load button in pixels
; 0 = auto (default: 110px scaled)
; Right edge is anchored at default position
; Example: LoadButtonWidth=80 (narrower button, extends left edge inward)
; Example: LoadButtonWidth=140 (wider button, extends left edge outward)
